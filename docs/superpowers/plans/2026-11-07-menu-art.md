# Painted Title Screen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The title screen becomes the painted menu art (`video/MENU.png`), with its five painted buttons acting as the real Inizia / Come si gioca / Impostazioni / Crediti / Esci buttons.

**Architecture:** `MainMenuScreen` stops building a logo plus text buttons and instead shows `assets/ui/menu.png` in an aspect-preserving, centred box. Five invisible `Button` hotspots are laid out as fractions of that box, so they track the painting at every window size. Hover/focus brightens the painted shape with an additive re-draw of the same texture, clipped to the hotspot region. Settings and credits panels are unchanged and open over the art.

**Tech Stack:** Godot 4.7.2, GDScript, Compatibility renderer, custom headless test runner (`tests/run_tests.gd`).

**Spec:** `docs/superpowers/specs/2026-11-07-menu-art-design.md`

## Global Constraints

- Player-facing text stays Italian (`MainMenuScreen.GAME_TITLE == "BalanSea"` must keep passing).

## Review Focus

- **Any window aspect** (ultra-wide, 16:9, portrait): the painting must stay centred and fully visible, and every hotspot must still sit on its painted pill. Pinned by the `art_box` tests (portrait + 16:9) in Task 2 and the 700×1100 screenshot in Task 4.
- **Zero/negative viewport** at first layout, before the tree gives the control its size: no NaN or inverted rects. Pinned by `test_art_box_is_empty_for_an_empty_viewport`.
- **Lighting up the right pixels**: hover/focus must brighten the pill, not the transparent background around it. Pinned by the screenshot in Task 4 (INIZIA glows on open).
- **Web build**: ESCI must disappear instead of looking clickable; the cover must eat clicks. Code-only (no web runtime locally): the cover is a child added after the hotspots and gated on `OS.has_feature("web")`.
- **Settings/credits round-trip**: pressing those painted buttons must hide the art, and Back must restore it with INIZIA focused. Verified by the existing flow plus the Task 4 screenshots.

  - `MenuHotspots.RECTS: Dictionary` — key (`String`) → `Rect2` in fractions of the art, in focus order: `"play"`, `"tutorial"`, `"settings"`, `"credits"`, `"exit"`.

  - `MenuHotspots.hotspot_rect(viewport: Vector2, key: String) -> Rect2` — the hotspot in screen coordinates.
  - `MenuHotspots.hotspot_region(key: String) -> Rect2` — the same rect in `ART_SIZE` texture pixels (for the additive glow region).

```gdscript
extends TestCase
## The painted title screen's hotspots must stay on the painting at any size.

func test_art_box_fits_16_by_9_without_cropping() -> void:
	var box := MenuHotspots.art_box(Vector2(1280, 720))
	assert_almost(box.size.y, 720.0)
	assert_almost(box.size.x, 720.0 * MenuHotspots.ART_SIZE.x / MenuHotspots.ART_SIZE.y)
	assert_almost(box.position, Vector2((1280.0 - box.size.x) * 0.5, 0.0))


func test_art_box_fits_a_portrait_viewport() -> void:
	var box := MenuHotspots.art_box(Vector2(700, 1100))
	assert_almost(box.size.x, 700.0)
	assert_almost(box.size.y, 700.0 * MenuHotspots.ART_SIZE.y / MenuHotspots.ART_SIZE.x)
	assert_almost(box.position.y, (1100.0 - box.size.y) * 0.5)


func test_art_box_is_empty_for_an_empty_viewport() -> void:
	assert_eq(MenuHotspots.art_box(Vector2.ZERO), Rect2())
	assert_eq(MenuHotspots.hotspot_rect(Vector2.ZERO, "play"), Rect2())


func test_every_hotspot_lies_inside_the_art() -> void:
	for key: String in MenuHotspots.RECTS:
		var rect: Rect2 = MenuHotspots.RECTS[key]
		assert_true(rect.position.x >= 0.0 and rect.position.y >= 0.0, key)
		assert_true(rect.end.x <= 1.0 and rect.end.y <= 1.0, key)
		assert_true(rect.size.x > 0.0 and rect.size.y > 0.0, key)


func test_hotspots_do_not_overlap() -> void:
	var keys := MenuHotspots.RECTS.keys()
	for i in keys.size():
		for j in range(i + 1, keys.size()):
			var a: Rect2 = MenuHotspots.RECTS[keys[i]]
			var b: Rect2 = MenuHotspots.RECTS[keys[j]]
			assert_false(a.intersects(b), "%s overlaps %s" % [keys[i], keys[j]])


func test_focus_order() -> void:
	assert_eq(MenuHotspots.RECTS.keys(),
		["play", "tutorial", "settings", "credits", "exit"])


func test_hotspot_rect_follows_the_art_box() -> void:
	var viewport := Vector2(1280, 720)
	var art := MenuHotspots.art_box(viewport)

	assert_almost(rect.size, art.size * fraction.size)


func test_hotspot_region_is_in_texture_pixels() -> void:
	var fraction: Rect2 = MenuHotspots.RECTS["play"]
	var region := MenuHotspots.hotspot_region("play")
	assert_almost(region.position, fraction.position * MenuHotspots.ART_SIZE)
	assert_almost(region.size, fraction.size * MenuHotspots.ART_SIZE)
```

Hotspot builder (one method, `_build_hotspot(key: String, handler: Callable) -> Button`):

```gdscript
func _build_hotspot(key: String, handler: Callable) -> Button:
	var button := Button.new()
	button.name = "Hotspot_" + key
	button.focus_mode = Control.FOCUS_ALL
	button.mouse_default_cursor_shape = Control.CURSOR_POINTING_HAND
	for item in ["normal", "hover", "pressed", "focus", "disabled"]:
		button.add_theme_stylebox_override(item, StyleBoxEmpty.new())
	button.pressed.connect(handler)
	# Additive re-draw of the painted pill, clipped to its region: brightens the
	# shape without touching the transparent background around it.
	var glow := TextureRect.new()
	glow.texture = MENU_TEXTURE
	glow.region_enabled = true
	glow.region_rect = MenuHotspots.hotspot_region(key)
	glow.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	glow.stretch_mode = TextureRect.STRETCH_SCALE
	glow.texture_filter = CanvasItem.TEXTURE_FILTER_LINEAR_WITH_MIPMAPS
	glow.material = _glow_material
	glow.modulate = Color(1, 1, 1, 0.22)
	glow.mouse_filter = Control.MOUSE_FILTER_IGNORE
	glow.visible = false
	button.add_child(glow)
	glow.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	button.mouse_entered.connect(func() -> void: glow.visible = true)
	button.mouse_exited.connect(func() -> void: glow.visible = false)
	button.focus_entered.connect(func() -> void: glow.visible = true)
	button.focus_exited.connect(func() -> void: glow.visible = false)
	button.button_down.connect(func() -> void: glow.modulate = Color(1, 1, 1, 0.4))
	button.button_up.connect(func() -> void: glow.modulate = Color(1, 1, 1, 0.22))
	return button
```

`_glow_material` is created once in `_build_menu_panel`: `CanvasItemMaterial.new()` with `blend_mode = CanvasItemMaterial.BLEND_MODE_ADD`.

Layout method:

```gdscript
## Fit the painting (and therefore its hotspots) into the window, centred.
func _layout_art() -> void:
	var art := MenuHotspots.art_box(size)
	_art_box.position = art.position
	_art_box.size = art.size
	for key: String in _hotspots:
		var rect := MenuHotspots.hotspot_rect(size, key)
		var button: Button = _hotspots[key]
		button.position = rect.position - art.position
		button.size = rect.size
	var exit_rect := MenuHotspots.hotspot_rect(size, "exit").grow(6.0)
	_exit_cover.position = exit_rect.position - art.position
	_exit_cover.size = exit_rect.size
```
